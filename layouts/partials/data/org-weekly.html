"value" $v "n" $c) }}{{ $cells = $cells | append (string $v) }}{{ $n = add $n 1 -}} {{- else -}} {{- $pts = $pts | append (dict "date" . "value" nil "n" 0) }}{{ $cells = $cells | append "null" -}} {{- end -}} {{- $nLatest = $c -}} {{- end -}} {{- $from := "" }}{{ $to := "" }}{{ with $labels }}{{ $from = index . 0 }}{{ $to = index .

{{- /* Sortable key for a version string ("1.27.1", "2.0.0rc1", "1.23.0-beta.0"): four zero-padded numeric parts, then "~" for a release or " " for a pre-release, so a release sorts after its pre-releases. Returns a string. */ -}} {{- $v := string . -}} {{- $core := "" -}}{{- with findRE `^[0-9]+(\.[0-9]+)*` $v 1 }}{{ $core = index . 0 }}{{ end -}} {{- $rest := strings.TrimPrefix $core $v -}} {{- $parts := split $core "." -}} {{- $key := "" -}} {{- range $i := seq 0 3 -}} {{- $n := 0 -}} {{- if lt $i (len $parts) }}{{ with index $parts $i }}{{ $n = int (replaceRE `^0+([0-9])` "$1" .) }}{{ end }}{{ end -}} {{- $key = printf "%s%06d." $key $n -}} {{- end -}} {{- return printf "%s%s" $key (cond (eq $rest "") "~" (printf " %s" $rest)) -}}

{{- /* Monthly sums of daily points (spec 8.1 History: "Downloads per month"). Call: partial "pages/overview/monthly.html" (dict "points" $s.points "anchor" $latest) Months run continuously up to the anchor's month (at most 24). The first month is dropped when the series starts after day 1. partialLast = the last month is the anchor's month and the anchor is not its last day. Every month except the anchor's month needs daily points on at least 90% of its days; months below that are "null" (left blank), like partial weekly bins, so missing snapshots never read as a drop. Returns {labels ("YYYY-MM"), cells (strings, "null" = no data), data (JSON string), partialLast, gaps (a month was blanked)}. */ -}} {{- $t := time.AsTime .anchor -}} {{- $pts := .points | default slice -}} {{- $labels := slice }}{{ $cells := slice }}{{ $partialLast := false }}{{ $gaps := false -}} {{- if $pts -}} {{- $by := dict }}{{ $cnt := dict -}} {{- range $pts -}} {{- $ym := substr .date 0 7 -}} {{- $by = merge $by (dict $ym (add (float (index $by $ym | default 0)) (float .value))) -}} {{- $cnt = merge $cnt (dict $ym (add (int (index $cnt $ym | default 0)) 1)) -}} {{- end -}} {{- $first := (index $pts 0).date -}} {{- $firstYm := substr $first 0 7 -}} {{- if ne (substr $first 8 2) "01" }}{{ $firstYm = ((time.AsTime (printf "%s-01" $firstYm)).AddDate 0 1 0).Format "2006-01" }}{{ end -}} {{- $endYm := $t.Format "2006-01" -}} {{- $t0 := time.AsTime (printf "%s-01" $endYm) -}} {{- range $i := seq 0 23 -}} {{- $ym := ($t0.AddDate 0 (int (sub $i 23)) 0).Format "2006-01" -}} {{- if ge $ym $firstYm -}} {{- $labels = $labels | append $ym -}} {{- $v := index $by $ym -}} {{- $cell := "null" }}{{ if ne $v nil }}{{ $cell = string $v }}{{ end -}} {{- if and (ne $v nil) (ne $ym $endYm) -}} {{- $dim := ((time.AsTime (printf "%s-01" $ym)).AddDate 0 1 -1).Day -}} {{- $n := int (index $cnt $ym | default 0) -}} {{- if lt (mul $n 10) (mul $dim 9) }}{{ $cell = "null" }}{{ $gaps = true }}{{ end -}} {{- end -}} {{- $cells = $cells | append $cell -}} {{- end -}} {{- end -}} {{- $partialLast = and (gt (len $labels) 0) (eq (index $labels (sub (len $labels) 1)) $endYm) (eq (($t.AddDate 0 0 1).Format "2006-01") $endYm) -}} {{- end -}} {{- return dict "labels" $labels "cells" $cells "data" (printf "[%s]" (delimit $cells ",")) "partialLast" $partialLast "gaps" $gaps -}}

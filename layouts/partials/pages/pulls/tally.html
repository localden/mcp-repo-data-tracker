{{- /* Row tally for a list of actionability states. Duplicate-cluster states count their cluster members (the rows that render). Input: slice of state dicts. Returns dict {n, auth, maint, sla}. */ -}} {{- $n := 0 }}{{ $auth := 0 }}{{ $maint := 0 }}{{ $sla := 0 -}} {{- range . -}} {{- $rows := .prs | default slice -}} {{- if and (eq .name "duplicate-cluster") .clusters -}} {{- $rows = slice -}} {{- range .clusters }}{{ range .members }}{{ $rows = $rows | append . }}{{ end }}{{ end -}} {{- end -}} {{- range $rows -}} {{- $n = add $n 1 -}} {{- if .is_auth }}{{ $auth = add $auth 1 }}{{ end -}} {{- if .is_maintainer }}{{ $maint = add $maint 1 }}{{ end -}} {{- if .sla_breach }}{{ $sla = add $sla 1 }}{{ end -}} {{- end -}} {{- end -}} {{- return dict "n" $n "auth" $auth "maint" $maint "sla" $sla -}}

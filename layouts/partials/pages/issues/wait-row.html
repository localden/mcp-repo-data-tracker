{{- /* One row of a "Waiting for a reply" list (design system v4 4.16). Ctx: dict {row, org, i}. row {w (days waiting), it (the metrics.json item: number, title, url, createdAt, labels, commentCount), name, slug}. org true adds the Repository cell and the repository slot of the item cell (shown below 64rem). i the row's index in the default sort, for the first-paint marks (data-over past 25, data-over-phone past 10). Cells in the fixed order: Repository, Issue, Comments, Waiting. Item meta slots: id, repository, priority, labels (these rows have no author in the data). Tokens: over7, repo:. */ -}} {{- $row := .row }}{{ $it := $row.it }}{{ $org := eq .org true }}{{ $i := int (.i | default 0) -}} {{- $prios := slice "P0" "P1" "P2" "P3" -}} {{- $prio := "" }}{{ $tags := slice -}} {{- range $it.labels | default slice -}} {{- if in $prios (string .) }}{{ if not $prio }}{{ $prio = string . }}{{ end }}{{ else }}{{ $tags = $tags | append (string .) }}{{ end -}} {{- end -}} {{- $tok := slice -}} {{- if ge $row.w 7.0 }}{{ $tok = $tok | append "over7" }}{{ end -}} {{- if $org }}{{ $tok = $tok | append (printf "repo:%s" $row.slug) }}{{ end -}} {{- $cc := int ($it.commentCount | default 0) -}} {{- if $org }} {{ $row.name }} {{- end }} {{ partial "ui/row-item.html" (dict "title" (string ($it.title | default "")) "href" $it.url "id" (printf "#%d" (int ($it.number | default 0))) "repo" (cond $org ($row.name | default "") "") "prio" $prio "tags" $tags) }} {{ partial "ui/value.html" (dict "kind" "count" "v" $cc) }} {{ partial "ui/value.html" (dict "kind" "dur" "d" $row.w "unit" "d" "compact" true "opened" $it.createdAt) }}

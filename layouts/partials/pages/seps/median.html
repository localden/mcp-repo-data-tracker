{{- /* Median of a slice of numbers (median age per stage, median time to final). Returns a float, or -1.0 for an empty slice. */ -}} {{- $v := sort (. | default slice) -}} {{- $n := len $v }}{{ $m := -1.0 -}} {{- if gt $n 0 -}} {{- $mid := div $n 2 -}} {{- if modBool $n 2 -}} {{- $m = div (add (float (index $v (sub $mid 1))) (float (index $v $mid))) 2.0 -}} {{- else -}} {{- $m = float (index $v $mid) -}} {{- end -}} {{- end -}} {{- return $m -}}

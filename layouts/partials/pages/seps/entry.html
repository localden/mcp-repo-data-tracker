", ") }}{{ end }}{{ end -}} {{- end -}} {{- end -}} {{- /* Waiting on maintainers (3.4-1): the first tile that matches, in tile order */ -}} {{- $needsSponsor := and (eq $stage "proposal") (eq $kind "none") -}} {{- $tile := "" -}} {{- if $needsSponsor }}{{ $tile = "t-sponsor" -}} {{- else if and $reviewKnown (not $reviewed) (not $notReady) (not $parked) (ne $stage "merged") }}{{ $tile = "t-review" -}} {{- else if eq $stage "in-review" }}{{ $tile = "t-decision" -}} {{- else if eq $stage "accepted" }}{{ $tile = "t-accepted" }}{{ end -}} {{- $tags := slice -}} {{- with $tile }}{{ $tags = $tags | append . }}{{ end -}} {{- if $needsSponsor }}{{ $tags = $tags | append "needs-sponsor" }}{{ end -}} {{- if and $reviewKnown (not $reviewed) }}{{ $tags = $tags | append "no-review" }}{{ end -}} {{- if $wg }}{{ $tags = $tags | append "wg" }}{{ end -}} {{- if $notReady }}{{ $tags = $tags | append "not-ready" }}{{ end -}} {{- if $parked }}{{ $tags = $tags | append "parked" }}{{ end -}} {{- $text := lower (delimit (slice $sep $title $author $sponsor (delimit $labels " ")) "|") -}} {{- return dict "num" $num "sep" $sep "sepNum" $sepNum "isPR" (eq $sepNum "") "title" $title "url" (string ($e.url | default "")) "author" $author "sponsor" $sponsor "sponsorKind" $kind "stage" $stage "age_h" $age_h "upd_h" $upd_h "stage_h" $stage_h "labels" $labels "wg" $wg "notReady" $notReady "reviewKnown" $reviewKnown "reviewed" $reviewed "coreKnown" $coreKnown "coreReviewed" $coreReviewed "needsLabel" (eq $e.needsLabel true) "sponsoredButProposal" (eq $e.sponsoredButProposal true) "reviewTitle" $reviewTitle "needsSponsor" $needsSponsor "tile" $tile "tags" $tags "text" $text -}}

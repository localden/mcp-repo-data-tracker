{{- /* One seps.json entry, normalised once for every module that shows SEPs (design system v4 4.16, 5.6). Returns a dict, never emits. Context: dict "e" (the seps.json entry) "stage" (one of "proposal", "draft", "in-review", "accepted", "parked", "merged") "ref" (Unix seconds of seps.json lastUpdated) "core" (lowercased core and lead maintainer logins). Keys: num (the PR number); sep: the id of the first meta slot, always in the one grammar "SEP-2028": the number the title names, else the PR number (a SEP is numbered by its PR), so one list never mixes "SEP-3392" and "#3440"; title: verbatim, minus the leading "SEP-2028: " prefix (the id is the first meta slot); url; author; createdAt; sponsor + sponsorKind: "handle", "self" (the sponsor is the author, or there is no sponsor and the author is a core or lead maintainer) or "none"; needsSponsor: a proposal with none; age_h, upd_h: hours (pages/seps/age.html), -1.0 when unknown; upd_h reads lastActivity* when the data carries it, else the time since the PR last changed; labels: GitHub labels verbatim for the label tags (stage labels are the group, wg-support-needed is the attribute tag); wg: carries wg-support-needed; tags: the data-facets tokens ("needs-sponsor", "wg"). */ -}} {{- $e := .e }}{{ $stage := .stage }}{{ $ref := .ref }}{{ $core := .core | default slice -}} {{- $num := int ($e.number | default 0) -}} {{- $raw := strings.TrimSpace (string ($e.title | default "")) -}} {{- $sepNum := "" -}} {{- with findRE `(?i)SEP-[0-9]+` $raw 1 }}{{ $sepNum = replaceRE `(?i)^SEP-` "" (index . 0) }}{{ end -}} {{- $sep := printf "SEP-%d" $num }}{{ with $sepNum }}{{ $sep = printf "SEP-%s" . }}{{ end -}} {{- $title := strings.TrimSpace (replaceRE `^\s*(?i:sep\(\s*SEP-[0-9]+\s*\)|SEP(-[0-9]+)?)\s*:\s*` "" $raw) -}} {{- if not $title }}{{ $title = $raw }}{{ end -}} {{- /* people: one sponsor rule, so the tile, the facet, the stage table and the Sponsor column agree */ -}} {{- $author := string ($e.author | default "") -}} {{- $sp := string ($e.sponsor | default "") -}} {{- $kind := "none" }}{{ $sponsor := "" -}} {{- if $sp -}} {{- if or (eq $sp "self") (eq (lower $sp) (lower $author)) }}{{ $kind = "self" }}{{ $sponsor = $author }}{{ else }}{{ $kind = "handle" }}{{ $sponsor = $sp }}{{ end -}} {{- else if and $author (in $core (lower $author)) -}} {{- $kind = "self" }}{{ $sponsor = $author -}} {{- end -}} {{- $needsSponsor := and (eq $stage "proposal") (eq $kind "none") -}} {{- /* durations in hours */ -}} {{- $age_h := partial "pages/seps/age.html" (dict "days" $e.daysWaiting "at" $e.createdAt "ref" $ref) -}} {{- $upd_h := -1.0 -}} {{- if or (ne $e.lastActivityDays nil) $e.lastActivityAt -}} {{- $upd_h = partial "pages/seps/age.html" (dict "days" $e.lastActivityDays "at" $e.lastActivityAt "ref" $ref) -}} {{- else -}} {{- $upd_h = partial "pages/seps/age.html" (dict "days" $e.daysInCurrentStatus "at" $e.updatedAt "ref" $ref) -}} {{- end -}} {{- /* label tags: GitHub labels verbatim; the stage labels are the group row and wg-support-needed is an attribute tag */ -}} {{- $skip := slice "sep" "proposal" "draft" "in-review" "in review" "accepted" "final" "wg-support-needed" -}} {{- $labels := slice }}{{ $wg := false -}} {{- range ($e.labels | default slice) -}} {{- $k := lower (string .) -}} {{- if eq $k "wg-support-needed" }}{{ $wg = true }}{{ end -}} {{- if not (in $skip $k) }}{{ $labels = $labels | append (string .) }}{{ end -}} {{- end -}} {{- $tags := slice -}} {{- if $needsSponsor }}{{ $tags = $tags | append "needs-sponsor" }}{{ end -}} {{- if $wg }}{{ $tags = $tags | append "wg" }}{{ end -}} {{- return dict "num" $num "sep" $sep "title" $title "url" (string ($e.url | default "")) "author" $author "createdAt" (string ($e.createdAt | default "")) "sponsor" $sponsor "sponsorKind" $kind "needsSponsor" $needsSponsor "stage" $stage "age_h" $age_h "upd_h" $upd_h "labels" $labels "wg" $wg "tags" $tags -}}

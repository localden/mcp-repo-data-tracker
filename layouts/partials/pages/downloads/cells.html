{{- /* Values for one SDK package (brief 3.5-1), shared by the packages table, its phone blocks and the small-multiple headers. A partial window is never shown as a full one. */ -}} {{- $value := "" }}{{ $flag := "" -}} {{- if and (not $p.short) $d.complete7 -}} {{- $value = partial "ui/num.html" (dict "v" $d.dl7 "mode" "compact") -}} {{- else if gt $d.days_covered 0 -}} {{- $value = printf `%s` (partial "ui/num.html" (dict "v" $d.sum7 "mode" "compact")) | safeHTML -}} {{- $flag = partial "ui/flag.html" (dict "text" (printf "%d of 7 days" $d.days_covered) "title" (printf "%s: %d of 7 complete days" $d.window $d.days_covered)) -}} {{- else if $d.complete_through -}} {{- $value = partial "ui/num.html" (dict "v" nil "na" (printf "no complete days in %s" $d.window)) -}} {{- $flag = partial "ui/flag.html" (dict "text" "0 of 7 days" "title" (printf "%s: no complete days" $d.window)) -}} {{- else -}} {{- $value = partial "ui/num.html" (dict "v" nil "na" "no data collected yet") -}} {{- end -}} {{- /* Changes: data/downloads.html sets delta7 / delta28 only when both 7-day windows are complete. */ -}} {{- $has7 := and (ne $d.delta7 nil) (ne $d.dl7 nil) (ne $d.prev7 nil) -}} {{- $chg7 := partial "ui/num.html" (dict "v" nil "na" "needs two complete 7-day windows") -}} {{- if $has7 }}{{ $chg7 = partial "ui/delta.html" (dict "cur" $d.dl7 "prev" $d.prev7 "unit" "percent" "polarity" "up") }}{{ end -}} {{- $has28 := and (ne $d.delta28 nil) (ne $d.dl7 nil) (ne $d.dl7_28 nil) -}} {{- $chg28 := partial "ui/num.html" (dict "v" nil "na" "needs complete 7-day windows 4 weeks apart") -}} {{- if $has28 }}{{ $chg28 = partial "ui/delta.html" (dict "cur" $d.dl7 "prev" $d.dl7_28 "unit" "percent" "polarity" "up") }}{{ end -}} {{- /* 12-week sparkline on the shared weeks: complete weekly sums only; a missing week breaks the line. */ -}} {{- $sc := newScratch -}} {{- range $d.weekly.points }}{{ if ne .value nil }}{{ $sc.SetInMap "w" .date .value }}{{ end }}{{ end -}} {{- $wm := $sc.Get "w" | default dict -}} {{- $pts := slice }}{{ $n := 0 }}{{ $lo := 0.0 }}{{ $hi := 0.0 -}} {{- range $w := $weeks -}} {{- $v := index $wm $w -}} {{- $pts = $pts | append (dict "date" $w "value" $v) -}} {{- if ne $v nil -}} {{- $f := float $v -}} {{- if or (eq $n 0) (lt $f $lo) }}{{ $lo = $f }}{{ end }}{{ if or (eq $n 0) (gt $f $hi) }}{{ $hi = $f }}{{ end }}{{ $n = add $n 1 -}} {{- end -}} {{- end -}} {{- $spark := "" -}} {{- if ge $n 4 -}} {{- $spark = printf `%sLowest week %s, highest week %s` (partial "ui/spark.html" (dict "points" $pts "size" "cell")) (partial "ui/num.html" (dict "v" $lo "mode" "compact" "plain" true)) (partial "ui/num.html" (dict "v" $hi "mode" "compact" "plain" true)) | safeHTML -}} {{- else if $weeks -}} {{- if not $d.first }}{{ $spark = partial "ui/num.html" (dict "v" nil "na" "no data collected yet") -}} {{- else if ge $d.first (index $weeks 0) }}{{ $spark = printf `Tracking since %s` (partial "ui/date.html" (dict "d" $d.first "style" "short")) | safeHTML -}} {{- else }}{{ $spark = `Too few complete weeks` | safeHTML }}{{ end -}} {{- else -}} {{- /* no package has a complete week yet, so there are no shared weeks: missing, not empty (6.1 rule 8) */ -}} {{- $spark = partial "ui/num.html" (dict "v" nil "na" "no complete weeks yet") -}} {{- end -}} {{- /* Lifetime: the registry's own lifetime total. npm has none; its running sum is labeled with its basis until D2-6 seeds it from Nov 2024 (downloads-daily.json › seeded_from), then "all time" like the others (brief 3.5-1, OV-27, DV-X1). */ -}} {{- $life := partial "ui/num.html" (dict "v" $d.lifetime "mode" "compact" "na" "no registry total") -}} {{- $lifeSub := "" -}} {{- if ne $d.lifetime nil -}} {{- $lifeSub = "all time" -}} {{- if eq $d.registry "npm" -}} {{- $seed := "" }}{{ with index $p.r.data "downloads-daily" }}{{ $seed = .seeded_from | default "" }}{{ end -}} {{- if or (eq $seed "") (gt $seed "2024-11-01") -}} {{- with $d.lifetime_since }}{{ $lifeSub = printf "since %s" (partial "ui/date.html" .) }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- $through := partial "ui/num.html" (dict "v" nil "na" "no data collected yet") -}} {{- with $d.complete_through -}} {{- $through = printf `%s` (partial "ui/date.html" .) (partial "ui/date.html" (dict "d" . "style" "short")) | safeHTML -}} {{- end -}} {{- return dict "value" $value "flag" $flag "chg7" $chg7 "chg28" $chg28 "has7" $has7 "has28" $has28 "spark" $spark "life" $life "lifeSub" $lifeSub "through" $through -}}

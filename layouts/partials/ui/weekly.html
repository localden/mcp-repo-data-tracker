{{- $t := time.AsTime .anchor -}}{{- $weeks := .weeks | default 40 -}}{{- $agg := .agg | default "last" -}} {{- $labels := slice }}{{ $cells := slice -}} {{- range $i := seq 0 (sub $weeks 1) -}} {{- $k := sub (sub $weeks 1) $i -}} {{- $end := ($t.AddDate 0 0 (int (mul -7 $k))).Format "2006-01-02" -}} {{- $start := ($t.AddDate 0 0 (int (sub (mul -7 $k) 7))).Format "2006-01-02" -}} {{- $n := 0 }}{{ $sum := 0.0 }}{{ $lastv := "null" -}} {{- range $.points -}}{{- if and (gt .date $start) (le .date $end) -}}{{ $n = add $n 1 }}{{ $sum = add $sum (float .value) }}{{ $lastv = string .value }}{{- end -}}{{- end -}} {{- $labels = $labels | append $end -}} {{- if eq $agg "sum" }}{{ $cells = $cells | append (cond (ge $n 7) (string $sum) "null") }}{{ else }}{{ $cells = $cells | append $lastv }}{{ end -}} {{- end -}} {{- return dict "labels" $labels "cells" $cells "data" (printf "[%s]" (delimit $cells ",")) -}}

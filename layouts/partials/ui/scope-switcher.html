{{- /* Scope switcher (design system v4 4.2, ruling 16): the H1 IS the switcher. Renders the title row's wrap: the heading holding only the button, and the menu as its sibling. ui/page-head.html calls it when it gets a scope; identical at every width. Dict {kind, repo, id}. kind: "issues" | "pulls" | "repo" (the page family whose sibling URLs it lists). repo: the current "owner/repo" or bare repo ("" is the org view, "All repositories"). id: element id prefix (default "page"): the H1 is "{id}-title", the menu "{id}-scope-menu". Menu rows: a column label row ("Repository" and "Open issues" or "Open PRs"), "All repositories" with the org total, a separator, then one link per repository with its open count (metrics.json open_count); the current row has the check. On /repos/** the same rows without the count column. */ -}} {{- $kind := .kind | default "repo" -}}{{- $cur := .repo | default "" -}} {{- $id := .id | default "page" -}} {{- $base := index (dict "issues" "issues" "pulls" "pulls" "repo" "repos") $kind | default "repos" -}} {{- $counted := ne $kind "repo" -}} {{- $rows := slice }}{{ $total := 0 }}{{ $any := false }}{{ $title := "All repositories" -}} {{- range site.Data.repos.repositories -}} {{- $r := partial "data/repo.html" . -}} {{- $on := or (eq $cur $r.key) (eq $cur $r.repo) -}} {{- if $on }}{{ $title = $r.name }}{{ end -}} {{- $row := dict "text" $r.name "href" (printf "%s/%s/" $base $r.repo | relURL) "current" $on -}} {{- if $counted -}} {{- with $r.data.metrics }}{{ with index . $kind }}{{ if reflect.IsMap . }}{{ if ne .open_count nil -}} {{- $row = merge $row (dict "count" .open_count) }}{{ $total = add $total (int .open_count) }}{{ $any = true -}} {{- end }}{{ end }}{{ end }}{{ end -}} {{- end -}} {{- $rows = $rows | append $row -}} {{- end -}} {{- $all := dict "text" "All repositories" "href" (printf "%s/" $base | relURL) "current" (eq $cur "") "sep" true -}} {{- if and $counted $any }}{{ $all = merge $all (dict "count" $total) }}{{ end -}} {{- $items := slice $all | append $rows -}} {{- $cols := slice -}} {{- if and $counted $any }}{{ $cols = slice "Repository" (cond (eq $kind "issues") "Open issues" "Open PRs") }}{{ end -}}

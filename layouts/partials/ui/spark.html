{{- /* Sparkline (design system v4 4.20). One height (24px), one colour (chart-1), never red, no area fill, no dashes, no text inside. It always emits one element: the trend, or the empty baseline. Dict {points, kind, size, reason, unit}. points weekly bins [{date, value | nil}] (data/week-bins.html, data/org-weekly.html). kind "level" (default): a 1.5px line with an end dot. "flow": one column per week, 2px apart; downloads are a flow everywhere. size "tile" (default): the full width of a metric tile's visual slot, plus a 16px row with the first and last week as dates. "cell": 80 by 24 in a v-col-spark column, no range row. reason "no-history": the measure has no stored series (the empty baseline, titled "No history for this measure"). unit how the wrapper title prints low and high: "count" (default), "volume" or "dur" (hours). Gaps: a full-height gap rectangle over missing weeks, leading and trailing runs included. Empty (fewer than four non-null points, or no history): a solid baseline. The drawing is aria-hidden; the wrapper title gives the window, low and high. */ -}} {{- $pts := .points | default slice -}} {{- $size := cond (in (slice "cell" "mini") (.size | default "tile")) "cell" "tile" -}} {{- $kind := .kind | default "level" -}} {{- if in (slice "column" "flow") $kind }}{{ $kind = "flow" }}{{ else }}{{ $kind = "level" }}{{ end -}} {{- $unit := .unit | default "count" -}} {{- $n := 0 }}{{ $lo := 0.0 }}{{ $hi := 0.0 -}} {{- range $pts -}} {{- if ne .value nil -}} {{- $f := float .value -}} {{- if or (eq $n 0) (lt $f $lo) }}{{ $lo = $f }}{{ end }}{{ if or (eq $n 0) (gt $f $hi) }}{{ $hi = $f }}{{ end }}{{ $n = add $n 1 -}} {{- end -}} {{- end -}} {{- $len := len $pts -}} {{- $empty := or (lt $n 4) (eq .reason "no-history") (lt $len 2) -}} {{- $first := "" }}{{ $last := "" -}} {{- if gt $len 0 }}{{ $first = (index $pts 0).date }}{{ $last = (index $pts (sub $len 1)).date }}{{ end -}} {{- $title := "Trend available after 4 weeks of data" -}} {{- if eq .reason "no-history" }}{{ $title = "No history for this measure" -}} {{- else if not $empty -}} {{- $fk := cond (eq $unit "volume") "fmt/volume.html" (cond (eq $unit "dur") "fmt/dur.html" "fmt/count.html") -}} {{- $loT := (partial $fk (cond (eq $unit "dur") (dict "h" $lo) (dict "v" $lo))).text -}} {{- $hiT := (partial $fk (cond (eq $unit "dur") (dict "h" $hi) (dict "v" $hi))).text -}} {{- $title = printf "%s, low %s, high %s" (partial "fmt/range.html" (dict "from" $first "to" $last)).text $loT $hiT -}} {{- end -}} {{- if $empty -}} {{- else if eq $kind "flow" -}} {{- range $pts -}} {{- if ne .value nil -}} {{- $pc := 0.0 }}{{ if gt $hi 0.0 }}{{ $pc = math.Max 4.0 (mul (div (math.Max 0.0 (float .value)) $hi) 100.0) }}{{ end -}} {{- else -}} {{- end -}} {{- end -}} {{- else -}} {{- /* level: x by index across the window, y inside a 3px inset; flat data reads flat (the span is at least 25% of the high, or 4) */ -}} {{- $h := 24.0 }}{{ $pad := 4.0 -}} {{- $minSpan := math.Max (mul 0.25 (math.Abs $hi)) (cond (eq $unit "dur") 1.0 4.0) -}} {{- if lt (sub $hi $lo) $minSpan -}} {{- $nonNeg := ge $lo 0.0 -}} {{- $mid := div (add $hi $lo) 2.0 }}{{ $lo = sub $mid (div $minSpan 2.0) }}{{ $hi = add $mid (div $minSpan 2.0) -}} {{- if and $nonNeg (lt $lo 0.0) }}{{ $hi = sub $hi $lo }}{{ $lo = 0.0 }}{{ end -}} {{- end -}} {{- $span := sub $hi $lo }}{{ $step := div 100.0 (float (sub $len 1)) }}{{ $half := div $step 2.0 -}} {{- $d := "" }}{{ $run := 0 }}{{ $lx := 0.0 }}{{ $ly := 0.0 }}{{ $gaps := slice }}{{ $gs := -1 }}{{ $dots := slice }}{{ $prev := "" -}} {{- range $i, $p := $pts -}} {{- if ne $p.value nil -}} {{- if ge $gs 0 }}{{ $gaps = $gaps | append (dict "a" $gs "b" (sub $i 1)) }}{{ $gs = -1 }}{{ end -}} {{- $x := mul (float $i) $step -}} {{- $y := sub (sub $h $pad) (mul (div (sub (float $p.value) $lo) $span) (sub $h (mul 2.0 $pad))) -}} {{- $pt := printf "%.2f,%.2f" $x $y -}} {{- if eq $run 0 }}{{ $d = printf "%sM%s" $d $pt }}{{ else }}{{ $d = printf "%s L%s" $d $pt }}{{ end -}} {{- $run = add $run 1 }}{{ $lx = $x }}{{ $ly = $y }}{{ $prev = $pt -}} {{- else -}} {{- if eq $run 1 }}{{ $dots = $dots | append $prev }}{{ end -}} {{- if lt $gs 0 }}{{ $gs = $i }}{{ end }}{{ $run = 0 -}} {{- end -}} {{- end -}} {{- if ge $gs 0 }}{{ $gaps = $gaps | append (dict "a" $gs "b" (sub $len 1)) }}{{ end -}} {{- $end := printf "M%.2f,%.2f l0.01,0" $lx $ly -}} {{- end -}} {{- if eq $size "tile" -}} {{ if and $first (not $empty) }}{{ partial "ui/value.html" (dict "kind" "date" "d" $first) }}{{ partial "ui/value.html" (dict "kind" "date" "d" $last) }}{{ end }} {{- end -}}
